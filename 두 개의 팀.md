# 코드
```
import sys
input = sys.stdin.readline

N = int(input())

V = [0] * N
child = [[] for _ in range(N)]

for i in range(N):
    V[i], p = map(int, input().split())
    if p != -1:
        child[p-1].append(i)

INF = float('inf')

dp = [0] * N
best1 = [-INF] * N
best2 = [-INF] * N

answer = -INF

for u in range(N-1, -1, -1):

    dp[u] = V[u]

    first = -INF
    second = -INF

    for v in child[u]:
        if dp[v] > 0:
            dp[u] += dp[v]
            candidate = best2[v]
        else:
            candidate = best1[v]

        best2[u] = max(best2[u], candidate)
        x = best1[v]

        if x > first:
            second = first
            first = x
        elif x > second:
            second = x

        best1[u] = max(best1[u], best1[v])

    best1[u] = max(best1[u], dp[u])

    if best2[u] != -INF:
        answer = max(answer, dp[u] + best2[u])

    if second != -INF:
        answer = max(answer, first + second)

print(answer)
```

    import sys
    N = int(sys.stdin.readline())
    C = []
    
    for _ in range(N):
        C.append(list(map(int, sys.stdin.readline().split(' '))))

    last_min = C[0][0]
    last_max = C[0][1]
    result = 1
    for i, j in C:
        if i > last_min and j < last_max:
            result += 1
        last_min = i
        last_max = j
    print(result)
# AI가 작성한 코드
    import sys
    
    input = sys.stdin.readline
    
    N = int(input())
    C = [list(map(int, input().split())) for _ in range(N)]
    
    C.sort()
    
    result = 0
    last_max = float('-inf')
    
    for i, j in C:
        if i > last_max:
            result += 1
            last_max = j
        else:
            last_max = min(last_max, j)

print(result)

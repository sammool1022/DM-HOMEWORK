    import sys
    N = int(sys.stdin.readline())
    C = []
    
    for _ in range(N):
        C.append(list(map(int, sys.stdin.readline().split(' '))))

    last_min = C[0][0]
    last_max = C[0][1]
    result = 1
    for i, j in C:
        if i > last_min and j < last_max:
            result += 1
        last_min = i
        last_max = j
    print(result)

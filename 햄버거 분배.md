    import sys
    N, k = map(int, sys.stdin.readline().split(' '))
    arr = list(sys.stdin.readline())[:N]
    result = 0
    for i in range(len(arr)):
        if arr[i] == 'P':
            start = max(i-k, 0)
            end = min(i+k+1, len(arr))
            a = arr[start:end].index('H') if 'H' in arr[start:end] else -1
            if a != -1:
                result += 1
                arr[start+a] = 'X'
    print(result)
